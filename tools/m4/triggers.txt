
TAI_ai_alt_path = {
	OR = {
		has_global_flag = TAI_ACCOMMODATION_FOCUS_PATH
		has_global_flag = TAI_ISLAND_FORTRESS_FOCUS_PATH
	}
}

TAI_ai_historical_path = {
	OR = {
		has_global_flag = TAI_HISTORICAL_FOCUS_PATH
		AND = {
			is_historical_focus_on = yes
			NOT = { TAI_ai_alt_path = yes }
		}
	}
}

TAI_ai_not_historical_path = {
	TAI_ai_alt_path = yes
}

TAI_ai_not_accommodation_path = {
	NOT = { has_global_flag = TAI_ACCOMMODATION_FOCUS_PATH }
	OR = {
		is_historical_focus_on = yes
		has_global_flag = TAI_HISTORICAL_FOCUS_PATH
		has_global_flag = TAI_ISLAND_FORTRESS_FOCUS_PATH
	}
}

TAI_ai_not_island_fortress_path = {
	NOT = { has_global_flag = TAI_ISLAND_FORTRESS_FOCUS_PATH }
	OR = {
		is_historical_focus_on = yes
		has_global_flag = TAI_HISTORICAL_FOCUS_PATH
		has_global_flag = TAI_ACCOMMODATION_FOCUS_PATH
	}
}
