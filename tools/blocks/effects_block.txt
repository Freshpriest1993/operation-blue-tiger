
# Operation Blue Tiger: party-army legacy is removed in two steps
TAI_reduce_party_army_legacy = {
	if = {
		limit = { has_idea = TAI_party_army }
		swap_ideas = {
			remove_idea = TAI_party_army
			add_idea = TAI_party_army_reduced
		}
	}
	else_if = {
		limit = { has_idea = TAI_party_army_reduced }
		remove_ideas = TAI_party_army_reduced
	}
}

TAI_remove_did_joint_penalty = {
	if = {
		limit = { has_idea = TAI_did_legacy }
		swap_ideas = {
			remove_idea = TAI_did_legacy
			add_idea = TAI_did_legacy_reformed
		}
	}
}

TAI_campaign_for_ruling_camp = {
	set_party_index_to_ruling_party = yes
	set_temp_variable = { party_popularity_increase = 0.02 }
	change_relative_party_popularity = yes
}
