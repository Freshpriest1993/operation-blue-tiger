
TAI_activate_reclamation_bop = {
	if = {
		limit = { check_variable = { TAI_taiwanese_identity > 59 } }
		set_power_balance = {
			id = TAI_reclamation_bop
			left_side = TAI_bop_restoration
			right_side = TAI_bop_self_determination
			set_value = 0.25
		}
	}
	else_if = {
		limit = { check_variable = { TAI_taiwanese_identity > 44 } }
		set_power_balance = {
			id = TAI_reclamation_bop
			left_side = TAI_bop_restoration
			right_side = TAI_bop_self_determination
			set_value = 0.10
		}
	}
	else_if = {
		limit = { check_variable = { TAI_taiwanese_identity > 29 } }
		set_power_balance = {
			id = TAI_reclamation_bop
			left_side = TAI_bop_restoration
			right_side = TAI_bop_self_determination
			set_value = -0.05
		}
	}
	else = {
		set_power_balance = {
			id = TAI_reclamation_bop
			left_side = TAI_bop_restoration
			right_side = TAI_bop_self_determination
			set_value = -0.20
		}
	}
}

TAI_update_reclamation = {
	if = {
		limit = {
			has_country_flag = TAI_reclamation_war
			NOT = { has_country_flag = TAI_mainland_falling }
			OR = {
				AND = {
					has_war_with = CHI
					CHI = { has_capitulated = yes }
				}
				AND = {
					NOT = { country_exists = CHI }
					ROC_holds_mainland = yes
				}
			}
		}
		set_country_flag = TAI_mainland_falling
		country_event = { id = reclamation.5 days = 1 }
	}
	else_if = {
		limit = {
			has_country_flag = TAI_reclamation_war
			NOT = { has_country_flag = TAI_mainland_falling }
			OR = {
				NOT = { country_exists = CHI }
				NOT = { has_war_with = CHI }
			}
		}
		clr_country_flag = TAI_reclamation_war
		country_event = { id = reclamation.18 days = 1 }
	}
}

ROC_init_administration = {
	set_country_flag = ROC_provisional_administration
	if = {
		limit = { NOT = { has_variable = ROC_regions_restored } }
		set_variable = { ROC_regions_restored = 0 }
	}
	if = {
		limit = { NOT = { has_variable = ROC_admin_reforms } }
		set_variable = { ROC_admin_reforms = 0 }
	}
	if = {
		limit = { NOT = { has_dynamic_modifier = { modifier = ROC_provisional_administration } } }
		add_dynamic_modifier = { modifier = ROC_provisional_administration }
	}
	ROC_update_administration = yes
}

ROC_update_administration = {
	set_temp_variable = { ROC_admin_points = ROC_regions_restored }
	add_to_temp_variable = { ROC_admin_points = ROC_admin_reforms }
	set_variable = { ROC_pa_factory = ROC_admin_points }
	multiply_variable = { ROC_pa_factory = 0.04 }
	subtract_from_variable = { ROC_pa_factory = 0.32 }
	clamp_variable = { var = ROC_pa_factory max = 0 }
	set_variable = { ROC_pa_stability = ROC_admin_points }
	multiply_variable = { ROC_pa_stability = 0.02 }
	subtract_from_variable = { ROC_pa_stability = 0.16 }
	clamp_variable = { var = ROC_pa_stability max = 0 }
	set_variable = { ROC_pa_pp = ROC_admin_points }
	multiply_variable = { ROC_pa_pp = 0.025 }
	subtract_from_variable = { ROC_pa_pp = 0.20 }
	clamp_variable = { var = ROC_pa_pp max = 0 }
	set_variable = { ROC_pa_conscription = ROC_admin_points }
	multiply_variable = { ROC_pa_conscription = 0.06 }
	subtract_from_variable = { ROC_pa_conscription = 0.48 }
	clamp_variable = { var = ROC_pa_conscription max = 0 }
}

ROC_end_administration = {
	clr_country_flag = ROC_provisional_administration
	if = {
		limit = { has_dynamic_modifier = { modifier = ROC_provisional_administration } }
		remove_dynamic_modifier = { modifier = ROC_provisional_administration }
	}
}

ROC_restore_region = {
	add_to_variable = { ROC_regions_restored = 1 }
	ROC_update_administration = yes
	add_stability = 0.02
	if = {
		limit = { check_variable = { ROC_regions_restored > 4 } }
		country_event = { id = reclamation.20 days = 1 }
	}
}

TAI_split_successor_states = {
	release = CHI
	every_state = {
		limit = {
			ROC_taiwan_island_state = yes
			is_owned_by = CHI
		}
		TAI = { transfer_state = PREV }
	}
	every_state = {
		limit = {
			ROC_taiwan_island_state = yes
			is_core_of = CHI
		}
		remove_core_of = CHI
	}
	set_capital = { state = 598 }
	ROC_end_administration = yes
	remove_power_balance = { id = TAI_reclamation_bop }
	set_country_flag = TAI_reclamation_concluded
	CHI = {
		set_variable = { ROC_regions_restored = TAI.ROC_regions_restored }
		if = {
			limit = { TAI = { has_country_flag = ROC_restored_north_china } }
			set_country_flag = ROC_restored_north_china
		}
		if = {
			limit = { TAI = { has_country_flag = ROC_restored_east_china } }
			set_country_flag = ROC_restored_east_china
		}
		if = {
			limit = { TAI = { has_country_flag = ROC_restored_south_china } }
			set_country_flag = ROC_restored_south_china
		}
		if = {
			limit = { TAI = { has_country_flag = ROC_restored_northeast } }
			set_country_flag = ROC_restored_northeast
		}
		if = {
			limit = { TAI = { has_country_flag = ROC_restored_northwest } }
			set_country_flag = ROC_restored_northwest
		}
		ROC_setup_mainland_republic = yes
		diplomatic_relation = {
			country = TAI
			relation = non_aggression_pact
			active = yes
		}
	}
}

TAI_renounce_mainland_claims = {
	every_state = {
		limit = {
			is_core_of = TAI
			NOT = { ROC_taiwan_island_state = yes }
		}
		remove_core_of = TAI
	}
	every_state = {
		limit = {
			is_claimed_by = TAI
			NOT = { ROC_taiwan_island_state = yes }
		}
		remove_claim_by = TAI
	}
}
