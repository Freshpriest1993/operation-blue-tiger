
TAI_reclamation_allowed = {
	custom_trigger_tooltip = {
		tooltip = TAI_reclamation_allowed_tt
		OR = {
			has_game_rule = { rule = TAI_return_to_mainland option = FREE }
			AND = {
				is_ai = no
				has_game_rule = { rule = TAI_return_to_mainland option = PLAYER_ONLY }
			}
		}
	}
}

TAI_beijing_in_crisis = {
	custom_trigger_tooltip = {
		tooltip = TAI_beijing_in_crisis_tt
		country_exists = CHI
		CHI = {
			OR = {
				has_civil_war = yes
				has_stability < 0.25
				any_enemy_country = {
					is_major = yes
					NOT = { original_tag = TAI }
				}
			}
		}
	}
}

		tooltip = TAI_repelled_invasion_tt
		all_owned_state = { is_controlled_by = ROOT }
	}
}
