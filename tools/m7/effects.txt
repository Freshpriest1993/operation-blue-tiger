
OBT_strait_update = {
	if = {
		limit = {
			country_exists = CHI
			country_exists = TAI
		}
		set_variable = { global.OBT_strait_fill = CHI.CHI_taiwan_strait_tension }
		clamp_variable = { var = global.OBT_strait_fill min = 0 max = 100 }
		if = {
			limit = { check_variable = { global.OBT_strait_fill > 79 } }
			set_temp_variable = { OBT_new_stage = 4 }
		}
		else_if = {
			limit = { check_variable = { global.OBT_strait_fill > 59 } }
			set_temp_variable = { OBT_new_stage = 3 }
		}
		else_if = {
			limit = { check_variable = { global.OBT_strait_fill > 39 } }
			set_temp_variable = { OBT_new_stage = 2 }
		}
		else_if = {
			limit = { check_variable = { global.OBT_strait_fill > 19 } }
			set_temp_variable = { OBT_new_stage = 1 }
		}
		else = {
			set_temp_variable = { OBT_new_stage = 0 }
		}
		if = {
			limit = {
				NOT = { has_global_flag = OBT_strait_initialized }
			}
			set_global_flag = OBT_strait_initialized
			set_variable = { global.OBT_strait_stage = OBT_new_stage }
			CHI = { OBT_strait_apply_stage = yes }
			TAI = { OBT_strait_apply_stage = yes }
			OBT_strait_init_snapshot = yes
		}
		else_if = {
			limit = {
				NOT = { check_variable = { global.OBT_strait_stage = OBT_new_stage } }
			}
			set_temp_variable = { OBT_old_stage = global.OBT_strait_stage }
			set_variable = { global.OBT_strait_stage = OBT_new_stage }
			CHI = { OBT_strait_apply_stage = yes }
			TAI = { OBT_strait_apply_stage = yes }
			if = {
				limit = {
					OR = {
						NOT = { has_global_flag = OBT_strait_event_cooldown }
						has_global_flag = { flag = OBT_strait_event_cooldown days > 179 }
					}
					NOT = { CHI = { has_war_with = TAI } }
				}
				clr_global_flag = OBT_strait_event_cooldown
				set_global_flag = OBT_strait_event_cooldown
				if = {
					limit = { check_variable = { OBT_new_stage > OBT_old_stage } }
					CHI = { country_event = { id = obt_strait.1 days = 1 } }
					TAI = { country_event = { id = obt_strait.3 days = 1 } }
				}
				else = {
					CHI = { country_event = { id = obt_strait.2 days = 1 } }
					TAI = { country_event = { id = obt_strait.4 days = 1 } }
				}
			}
		}
	}
}

OBT_strait_snapshot = {
	if = {
		limit = {
			country_exists = CHI
			country_exists = TAI
		}
		set_variable = { global.OBT_d_trust = CHI.CHI_TAI_trust }
		subtract_from_variable = { global.OBT_d_trust = global.OBT_prev_trust }
		set_variable = { global.OBT_prev_trust = CHI.CHI_TAI_trust }
		set_variable = { global.OBT_d_ussupport = CHI.CHI_taiwan_us_support }
		subtract_from_variable = { global.OBT_d_ussupport = global.OBT_prev_ussupport }
		set_variable = { global.OBT_prev_ussupport = CHI.CHI_taiwan_us_support }
		set_variable = { global.OBT_d_burden = TAI.TAI_blockade_burden }
		subtract_from_variable = { global.OBT_d_burden = global.OBT_prev_burden }
		set_variable = { global.OBT_prev_burden = TAI.TAI_blockade_burden }
		set_variable = { global.OBT_d_ptn = TAI.TAI_pressure_to_negotiate }
		subtract_from_variable = { global.OBT_d_ptn = global.OBT_prev_ptn }
		set_variable = { global.OBT_prev_ptn = TAI.TAI_pressure_to_negotiate }
		set_variable = { global.OBT_d_usconf = TAI.TAI_us_confidence }
		subtract_from_variable = { global.OBT_d_usconf = global.OBT_prev_usconf }
		set_variable = { global.OBT_prev_usconf = TAI.TAI_us_confidence }
		set_variable = { global.OBT_d_identity = TAI.TAI_taiwanese_identity }
		subtract_from_variable = { global.OBT_d_identity = global.OBT_prev_identity }
		set_variable = { global.OBT_prev_identity = TAI.TAI_taiwanese_identity }
	}
}

OBT_strait_init_snapshot = {
	if = {
		limit = {
			country_exists = CHI
			country_exists = TAI
		}
		set_variable = { global.OBT_prev_trust = CHI.CHI_TAI_trust }
		set_variable = { global.OBT_d_trust = 0 }
		set_variable = { global.OBT_prev_ussupport = CHI.CHI_taiwan_us_support }
		set_variable = { global.OBT_d_ussupport = 0 }
		set_variable = { global.OBT_prev_burden = TAI.TAI_blockade_burden }
		set_variable = { global.OBT_d_burden = 0 }
		set_variable = { global.OBT_prev_ptn = TAI.TAI_pressure_to_negotiate }
		set_variable = { global.OBT_d_ptn = 0 }
		set_variable = { global.OBT_prev_usconf = TAI.TAI_us_confidence }
		set_variable = { global.OBT_d_usconf = 0 }
		set_variable = { global.OBT_prev_identity = TAI.TAI_taiwanese_identity }
		set_variable = { global.OBT_d_identity = 0 }
	}
}

OBT_strait_apply_stage = {
	if = {
		limit = { original_tag = CHI }
		set_variable = { OBT_ss_stab = 0 }
		set_variable = { OBT_ss_pp = 0 }
		set_variable = { OBT_ss_trade = 0 }
		set_variable = { OBT_ss_factory = 0 }
		set_variable = { OBT_ss_cg = 0 }
		set_variable = { OBT_ss_org = 0 }
		set_variable = { OBT_ss_armyxp = 0 }
		set_variable = { OBT_ss_navyxp = 0 }
		if = {
			limit = { check_variable = { global.OBT_strait_stage = 0 } }
			set_variable = { OBT_ss_stab = 0.02 }
			set_variable = { OBT_ss_pp = 0.05 }
			set_variable = { OBT_ss_trade = 0.05 }
			set_variable = { OBT_ss_factory = 0.02 }
		}
		if = {
			limit = { check_variable = { global.OBT_strait_stage = 1 } }
			set_variable = { OBT_ss_pp = 0.01 }
			set_variable = { OBT_ss_trade = -0.02 }
		}
		if = {
			limit = { check_variable = { global.OBT_strait_stage = 2 } }
			set_variable = { OBT_ss_stab = -0.01 }
			set_variable = { OBT_ss_pp = -0.02 }
			set_variable = { OBT_ss_trade = -0.05 }
			set_variable = { OBT_ss_org = 0.02 }
			set_variable = { OBT_ss_armyxp = 0.05 }
			set_variable = { OBT_ss_navyxp = 0.05 }
		}
		if = {
			limit = { check_variable = { global.OBT_strait_stage = 3 } }
			set_variable = { OBT_ss_stab = -0.03 }
			set_variable = { OBT_ss_pp = -0.05 }
			set_variable = { OBT_ss_trade = -0.1 }
			set_variable = { OBT_ss_cg = 0.02 }
			set_variable = { OBT_ss_org = 0.03 }
			set_variable = { OBT_ss_armyxp = 0.1 }
			set_variable = { OBT_ss_navyxp = 0.1 }
		}
		if = {
			limit = { check_variable = { global.OBT_strait_stage = 4 } }
			set_variable = { OBT_ss_stab = -0.05 }
			set_variable = { OBT_ss_pp = -0.1 }
			set_variable = { OBT_ss_trade = -0.15 }
			set_variable = { OBT_ss_cg = 0.04 }
			set_variable = { OBT_ss_org = 0.05 }
			set_variable = { OBT_ss_armyxp = 0.15 }
			set_variable = { OBT_ss_navyxp = 0.15 }
		}
		if = {
			limit = { NOT = { has_dynamic_modifier = { modifier = OBT_strait_stage_CHI } } }
			add_dynamic_modifier = { modifier = OBT_strait_stage_CHI }
		}
	}
	if = {
		limit = { original_tag = TAI }
		set_variable = { OBT_ss_stab = 0 }
		set_variable = { OBT_ss_pp = 0 }
		set_variable = { OBT_ss_trade = 0 }
		set_variable = { OBT_ss_factory = 0 }
		set_variable = { OBT_ss_cg = 0 }
		set_variable = { OBT_ss_ws = 0 }
		set_variable = { OBT_ss_def = 0 }
		set_variable = { OBT_ss_consc = 0 }
		set_variable = { OBT_ss_aa = 0 }
		if = {
			limit = { check_variable = { global.OBT_strait_stage = 0 } }
			set_variable = { OBT_ss_stab = 0.02 }
			set_variable = { OBT_ss_pp = 0.05 }
			set_variable = { OBT_ss_trade = 0.05 }
			set_variable = { OBT_ss_factory = 0.02 }
		}
		if = {
			limit = { check_variable = { global.OBT_strait_stage = 1 } }
			set_variable = { OBT_ss_trade = -0.02 }
			set_variable = { OBT_ss_ws = 0.02 }
		}
		if = {
			limit = { check_variable = { global.OBT_strait_stage = 2 } }
			set_variable = { OBT_ss_pp = -0.02 }
			set_variable = { OBT_ss_trade = -0.03 }
			set_variable = { OBT_ss_ws = 0.03 }
			set_variable = { OBT_ss_def = 0.02 }
			set_variable = { OBT_ss_consc = 0.02 }
			set_variable = { OBT_ss_aa = 0.05 }
		}
		if = {
			limit = { check_variable = { global.OBT_strait_stage = 3 } }
			set_variable = { OBT_ss_stab = -0.02 }
			set_variable = { OBT_ss_pp = -0.04 }
			set_variable = { OBT_ss_trade = -0.06 }
			set_variable = { OBT_ss_cg = 0.02 }
			set_variable = { OBT_ss_ws = 0.05 }
			set_variable = { OBT_ss_def = 0.04 }
			set_variable = { OBT_ss_consc = 0.04 }
			set_variable = { OBT_ss_aa = 0.1 }
		}
		if = {
			limit = { check_variable = { global.OBT_strait_stage = 4 } }
			set_variable = { OBT_ss_stab = -0.04 }
			set_variable = { OBT_ss_pp = -0.06 }
			set_variable = { OBT_ss_trade = -0.1 }
			set_variable = { OBT_ss_cg = 0.04 }
			set_variable = { OBT_ss_ws = 0.08 }
			set_variable = { OBT_ss_def = 0.05 }
			set_variable = { OBT_ss_consc = 0.06 }
			set_variable = { OBT_ss_aa = 0.15 }
		}
		if = {
			limit = { NOT = { has_dynamic_modifier = { modifier = OBT_strait_stage_TAI } } }
			add_dynamic_modifier = { modifier = OBT_strait_stage_TAI }
		}
	}
}
